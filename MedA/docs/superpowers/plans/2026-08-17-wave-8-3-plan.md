```python
"""stats_evidence.py: pure math. No DB, no imports beyond Python stdlib (math only)."""
from __future__ import annotations
import math

def normal_cdf(x: float) -> float:
    if x <= -37: return 0.0
    if x >= 37: return 1.0
    return 0.5 * math.erfc(-x / math.sqrt(2))

def _p_low_gamma_series(a: float, x: float, max_iter: int = 200, eps: float = 1e-12) -> float:
    """Regularized lower incomplete gamma P(a,x) = γ(a,x)/Γ(a), via series (x small / moderate)."""
    if x < 0 or a <= 0:
        raise ValueError(f"invalid a={a}, x={x}")
    if x == 0:
        return 0.0
    # use log-gamma stable
    log_ga = math.lgamma(a)
    prefactor = math.exp(a * math.log(x) - x - log_ga)
    term = 1.0 / a   # n=0
    total = term
    for n in range(1, max_iter+1):
        term *= x / (a + n)
        total += term
        if abs(term) < eps * abs(total):
            break
    return prefactor * total

def chi2_cdf(x: float, df: int) -> float:
    if df <= 0:
        raise ValueError(f"df={df}")
    if x <= 0:
        return 0.0
    return _p_low_gamma_series(df / 2.0, x / 2.0)

def cohen_kappa(confusion_matrix: list[list[int | float]]) -> float:
    """Confusion matrix rows = reviewer A, cols = reviewer B, any size NxN. Simple Kappa, no weights."""
    N = sum(sum(row) for row in confusion_matrix)
    if N <= 0: return float("nan")
    n = len(confusion_matrix)
    po = sum(confusion_matrix[i][i] for i in range(n)) / N
    pe = 0.0
    for i in range(n):
        row_total = sum(confusion_matrix[i][j] for j in range(n))
        col_total = sum(confusion_matrix[j][i] for j in range(n))
        pe += row_total * col_total
    pe /= (N * N)
    if (1 - pe) == 0:
        return float("nan")
    return (po - pe) / (1 - pe)

# ===== binary effect sizes (S2, S3, S4) =====
def _apply_cc(a: float, n1: float, c: float, n2: float) -> tuple[float,float,float,float]:
    cc = 0.5
    b = n1 - a; d = n2 - c
    if min(a,b,c,d) == 0:
        return a+cc, n1+2*cc, c+cc, n2+2*cc   # classic 0.5 to all cells
    return a,n1,c,n2

def binary_rr_95ci(a: float, n1: float, c: float, n2: float, *, continuity_correction: bool = True):
    if continuity_correction:
        a,n1,c,n2 = _apply_cc(a,n1,c,n2)
    p1 = a/n1; p2 = c/n2
    if p1 <= 0 or p2 <= 0:
        # cannot compute log ratio; return 1.0 CI very wide
        return 1.0, float("nan"), float("nan"), 0.0, 1.0
    import math
    log_rr = math.log(p1/p2)
    se = math.sqrt(1/a - 1/n1 + 1/c - 1/n2)
    ci_low = math.exp(log_rr - 1.96 * se)
    ci_high = math.exp(log_rr + 1.96 * se)
    z = log_rr / se if se > 0 else 0.0
    p = 2 * (1 - normal_cdf(abs(z)))
    return math.exp(log_rr), ci_low, ci_high, z, p

def binary_or_95ci(a: float, n1: float, c: float, n2: float, *, continuity_correction: bool = True):
    if continuity_correction:
        a,n1,c,n2 = _apply_cc(a,n1,c,n2)
    b = n1-a; d = n2-c
    import math
    if min(a,b,c,d) <= 0:
        return float("nan"), float("nan"), float("nan"), 0.0, 1.0
    log_or = math.log((a*d)/(b*c))
    se = math.sqrt(1/a + 1/b + 1/c + 1/d)
    ci_low = math.exp(log_or - 1.96*se)
    ci_high = math.exp(log_or + 1.96*se)
    z = log_or/se if se > 0 else 0.0
    p = 2 * (1 - normal_cdf(abs(z)))
    return math.exp(log_or), ci_low, ci_high, z, p

def binary_rd_95ci(a: float, n1: float, c: float, n2: float):
    rd = a/n1 - c/n2
    se = math.sqrt(a*(n1-a)/(n1**3) + c*(n2-c)/(n2**3)) if n1>0 and n2>0 else 0.0
    ci_low = rd - 1.96*se; ci_high = rd + 1.96*se
    z = rd/se if se > 0 else 0.0
    p = 2 * (1 - normal_cdf(abs(z)))
    return rd, ci_low, ci_high, z, p

# ===== continuous S5 =====
def continuous_md_95ci(m1: float, s1: float, n1: int, m2: float, s2: float, n2: int):
    if n1 < 2 or n2 < 2 or s1 < 0 or s2 < 0:
        return float("nan"), float("nan"), float("nan"), 0.0, 1.0
    sp2 = ((n1-1)*s1*s1 + (n2-1)*s2*s2) / (n1+n2-2)
    md = m1 - m2
    se = math.sqrt(sp2 * (1/n1 + 1/n2))
    ci_low = md - 1.96*se; ci_high = md + 1.96*se
    z = md/se if se > 0 else 0.0
    p = 2 * (1 - normal_cdf(abs(z)))
    return md, ci_low, ci_high, z, p

# ===== Fixed IV S6, MH S7, DL S8 =====
def fixed_iv_pooled(studies: list[tuple], measure: str = "RR", model: str = "fixed_iv"):
    """studies = [(a,n1,c,n2), ...] for binary; or [((m1,s1,n1),(m2,s2,n2)),...] for continuous (tuple of 2 tuples).
       Returns pooled_effect, ci_low, ci_high, z, p
    """
    import math
    theta_i_ses = []
    for s in studies:
        if len(s) == 4 and all(isinstance(x, (int,float)) for x in s):
            a,n1,c,n2 = s
            if measure == "RR":
                eff,ci_l,ci_h,z,p = binary_rr_95ci(a,n1,c,n2)
                logeff = math.log(eff) if eff > 0 else 0.0
                se = (math.log(ci_h) - math.log(ci_l)) / (2*1.96) if ci_h>0 and ci_l>0 else float("inf")
                theta_i_ses.append( (logeff, se) )
            elif measure == "OR":
                eff,ci_l,ci_h,z,p = binary_or_95ci(a,n1,c,n2)
                logeff = math.log(eff) if eff > 0 else 0.0
                se = (math.log(ci_h) - math.log(ci_l)) / (2*1.96) if ci_h>0 and ci_l>0 else float("inf")
                theta_i_ses.append( (logeff, se) )
            elif measure == "RD":
                rd,ci_l,ci_h,z,p = binary_rd_95ci(a,n1,c,n2)
                se = (ci_h - ci_l) / (2*1.96)
                theta_i_ses.append( (rd, se) )
            else:
                raise ValueError(measure)
        else:
            # continuous ((m1,s1,n1),(m2,s2,n2))
            (m1,s1,n1),(m2,s2,n2) = s
            md,ci_l,ci_h,z,p = continuous_md_95ci(m1,s1,n1,m2,s2,n2)
            se = (ci_h - ci_l) / (2*1.96)
            theta_i_ses.append( (md, se) )

    sw = 0.0; swt = 0.0
    for theta, se in theta_i_ses:
        if se <= 0 or not math.isfinite(se):
            continue
        w = 1.0 / (se*se)
        sw += w; swt += w*theta
    if sw == 0:
        return float("nan"), float("nan"), float("nan"), 0.0, 1.0
    pooled = swt / sw
    se_pool = 1.0 / math.sqrt(sw)
    # differentiate: if log scale then exp, if MD then leave as-is
    is_log = measure in ("RR","OR")
    eff_pool = math.exp(pooled) if is_log else pooled
    ci_low = math.exp(pooled - 1.96*se_pool) if is_log else (pooled - 1.96*se_pool)
    ci_high = math.exp(pooled + 1.96*se_pool) if is_log else (pooled + 1.96*se_pool)
    z = pooled/se_pool if se_pool > 0 else 0.0
    p = 2 * (1 - normal_cdf(abs(z)))
    return eff_pool, ci_low, ci_high, z, p

def fixed_mh_pooled_rr(studies: list[tuple]):
    """studies = [(a,n1,c,n2),...] Mantel-Haenszel pooled RR, Robins-Breslow-Greenland SE."""
    num = 0.0; den = 0.0
    R_sum = 0.0; S_sum = 0.0; V_sum = 0.0
    for a,n1,c,n2 in studies:
        N = n1 + n2
        num += (a * n2) / N
        den += (c * n1) / N
        R = (a + c) / N
        S = (a * n2)/N * R
        T = (c * n1)/N * R
        U = (a * n2)/N * (1 - R) + (c * n1)/N * (1 - R)
        R_sum += S
        S_sum += T
        V_sum += U
    if den <= 0:
        return float("nan"), float("nan"), float("nan"), 0.0, 1.0
    import math
    rr_mh = num / den
    # Robins 1986 variance of log(RR_MH)
    var_log_rr = (R_sum + V_sum/2) / (2 * num * num) + (S_sum + V_sum/2) / (2 * den * den)
    if var_log_rr <= 0:
        return rr_mh, float("nan"), float("nan"), 0.0, 1.0
    se = math.sqrt(var_log_rr)
    ci_low = math.exp(math.log(rr_mh) - 1.96*se)
    ci_high = math.exp(math.log(rr_mh) + 1.96*se)
    z = math.log(rr_mh)/se
    p = 2*(1-normal_cdf(abs(z)))
    return rr_mh, ci_low, ci_high, z, p

def dl_random_pooled(ys_vs: list[tuple[float,float]]):
    """(yi, vi) where vi = SE^2. Returns pooled_effect(back-transformed?), ci, se, tau2."""
    import math
    # Step 1: IV fixed to compute Q and theta_fixed
    wi = [1.0/vi if vi > 0 else 0.0 for _,vi in ys_vs]
    K = len(ys_vs)
    sw = sum(wi); swt = sum(wi[i]*ys_vs[i][0] for i in range(K))
    if sw <= 0 or K < 2:
        return float("nan"), float("nan"), float("nan"), 0.0
    theta_f = swt / sw
    Q = sum(wi[i] * (ys_vs[i][0] - theta_f)**2 for i in range(K))
    # DL tau²
    denom = sw - sum(wi[i]*wi[i] for i in range(K)) / sw
    tau2 = max(0.0, (Q - (K-1)) / denom) if denom > 0 else 0.0
    w2 = [1.0 / (vi + tau2) if (vi + tau2) > 0 else 0.0 for _,vi in ys_vs]
    sw2 = sum(w2); sw2t = sum(w2[i]*ys_vs[i][0] for i in range(K))
    if sw2 <= 0:
        return float("nan"), float("nan"), float("nan"), tau2
    pooled = sw2t / sw2
    se = 1.0 / math.sqrt(sw2)
    return pooled, (pooled - 1.96*se, pooled + 1.96*se), se, tau2

def compute_heterogeneity(studies: list, measure: str = "RR") -> dict:
    """Compute Q, df, p_Q, I² for studies list."""
    import math
    # Re-use IV fixed log scale calculation
    theta_ses = []
    for s in studies:
        if len(s)==4:
            a,n1,c,n2 = s
            if measure == "RR":
                eff,ci_l,ci_h,_,_ = binary_rr_95ci(a,n1,c,n2)
                if eff > 0 and ci_l>0 and ci_h>0:
                    theta_ses.append( (math.log(eff), (math.log(ci_h)-math.log(ci_l))/(2*1.96)) )
            elif measure == "OR":
                eff,ci_l,ci_h,_,_ = binary_or_95ci(a,n1,c,n2)
                if eff > 0 and ci_l>0 and ci_h>0:
                    theta_ses.append( (math.log(eff), (math.log(ci_h)-math.log(ci_l))/(2*1.96)) )
    wi = [1.0/(se*se) if se>0 else 0.0 for _,se in theta_ses]
    K = len(theta_ses)
    df = max(0, K-1)
    sw = sum(wi)
    if sw <= 0 or K < 2:
        return {"Q":0.0,"df_Q":df,"p_Q":1.0,"I2_pct":0.0,"K":K}
    theta_pooled = sum(wi[i]*theta_ses[i][0] for i in range(K)) / sw
    Q = sum(wi[i]*(theta_ses[i][0] - theta_pooled)**2 for i in range(K))
    I2 = max(0.0, (Q - df)/Q * 100) if Q>0 else 0.0
    p_Q = 1.0 - chi2_cdf(Q, df)
    return {"Q":Q,"df_Q":df,"p_Q":p_Q,"I2_pct":I2,"K":K}
```
