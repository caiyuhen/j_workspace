# apps/agent-core/app/services/grade_engine.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Literal

DOMAIN_LEVEL = Literal["no_concerns", "some_concerns", "major_concerns"]
CERTAINTY   = Literal["High", "Moderate", "Low", "VeryLow"]
START_LEVEL = Literal["High", "Moderate", "Low", "VeryLow"]

DOMAIN_SCORE: dict[DOMAIN_LEVEL, int] = {
    "no_concerns": 0,
    "some_concerns": 1,
    "major_concerns": 2,
}

DOMAIN_KEYS_5: tuple[str, str, str, str, str] = (
    "risk_of_bias", "indirectness", "inconsistency", "imprecision", "publication_bias"
)

DOWNGRADE_TABLE_STR: dict[START_LEVEL, list[CERTAINTY]] = {
    #                       0           1            2             3+
    "High":     ["High",   "Moderate", "Low",       "VeryLow"],
    "Moderate": ["Moderate","Low",     "VeryLow",   "VeryLow"],
    "Low":      ["Low",    "VeryLow",  "VeryLow",   "VeryLow"],
    "VeryLow":  ["VeryLow","VeryLow",  "VeryLow",   "VeryLow"],
}

UPGRADE_TABLE_STR: dict[CERTAINTY, list[CERTAINTY]] = {
    #                      0            1            2            3
    "VeryLow":  ["VeryLow", "Low",      "Moderate",  "High"],
    "Low":      ["Low",     "Moderate", "High",      "High"],
    "Moderate": ["Moderate","High",     "High",      "High"],
    "High":     ["High",    "High",     "High",      "High"],
}

@dataclass(frozen=True, slots=True)
class Grade5Domains:
    risk_of_bias: DOMAIN_LEVEL
    indirectness: DOMAIN_LEVEL
    inconsistency: DOMAIN_LEVEL
    imprecision: DOMAIN_LEVEL
    publication_bias: DOMAIN_LEVEL

    def items(self):
        return (
            ("risk_of_bias", self.risk_of_bias),
            ("indirectness", self.indirectness),
            ("inconsistency", self.inconsistency),
            ("imprecision", self.imprecision),
            ("publication_bias", self.publication_bias),
        )

    def total_downgrade_score(self) -> int:
        return sum(DOMAIN_SCORE[v] for _, v in self.items())

@dataclass(frozen=True, slots=True)
class Grade3Upgrades:
    large_effect: bool
    dose_response: bool
    confounders_reduce: bool

    def total_upgrade_count(self) -> int:
        return (1 if self.large_effect else 0) + (1 if self.dose_response else 0) + (1 if self.confounders_reduce else 0)

def _validate_domains(d: Grade5Domains) -> None:
    # RULE O3 + O8 double-check (caller already validates; this is safe net)
    for _, v in d.items():
        if v not in DOMAIN_SCORE:
            raise GradeEngineError("grade_invalid_domain_value_not_in_3_level")
    # Already dataclass slots exact 5 keys, so O8 inherently satisfied.

class GradeEngineError(Exception):
    pass

def compute_certainty_final(
    domains: Grade5Domains,
    upgrades: Grade3Upgrades,
    start: START_LEVEL = "High",
) -> CERTAINTY:
    _validate_domains(domains)
    td = domains.total_downgrade_score()  # 0..10
    bucket_td = min(td, 3)  # table covers 0/1/2/3+
    after_td: CERTAINTY = DOWNGRADE_TABLE_STR[start][bucket_td]
    tu = upgrades.total_upgrade_count()  # 0..3
    bucket_tu = min(tu, 3)
    final: CERTAINTY = UPGRADE_TABLE_STR[after_td][bucket_tu]
    return final
```

# apps/agent-core/app/services/sof_table_engine.py
from __future__ import annotations
from dataclasses import dataclass
from pydantic import BaseModel
from app.services.grade_engine import Grade5Domains, Grade3Upgrades, CERTAINTY

@dataclass(frozen=True, slots=True)
class SofInputMeta:
    project_id: int
    outcome_id: int
    outcome_label: str
    participants_n: int
    studies_k: int
    effect_measure_label: str
    pooled_rr: float | None = None
    risk_control_baseline: float = 0.0

class SofRow12(BaseModel):
    project_id: int
    outcome_id: int
    outcome_label: str
    participants_n: int
    studies_k: int
    effect_measure_label: str
    risk_of_bias: str
    indirectness: str
    inconsistency: str
    imprecision: str
    publication_bias: str
    certainty: CERTAINTY
    absolute_risk_intervention: str | None = None
    absolute_risk_control: str | None = None
    comments: str | None = None

def _absolute_risk(meta: SofInputMeta) -> tuple[str, str]:
    """Baseline control risk, and predicted intervention risk via pooled RR."""
    rc = max(0.0, min(1.0, float(meta.risk_control_baseline)))
    if meta.pooled_rr is None:
        ri_str = "NR"
    else:
        ri = max(0.0, min(1.0, rc * float(meta.pooled_rr)))
        ri_str = f"{ri*100:.1f}%"
    rc_str = f"{rc*100:.1f}%"
    return ri_str, rc_str

def build_sof_row_12cols(
    meta: SofInputMeta,
    domains: Grade5Domains,
    upgrades: Grade3Upgrades,
    cer: CERTAINTY,
    comments: str | None = None,
) -> SofRow12:
    ri, rc = _absolute_risk(meta)
    return SofRow12(
        project_id=meta.project_id,
        outcome_id=meta.outcome_id,
        outcome_label=meta.outcome_label,
        participants_n=meta.participants_n,
        studies_k=meta.studies_k,
        effect_measure_label=meta.effect_measure_label,
        risk_of_bias=domains.risk_of_bias,
        indirectness=domains.indirectness,
        inconsistency=domains.inconsistency,
        imprecision=domains.imprecision,
        publication_bias=domains.publication_bias,
        certainty=cer,
        absolute_risk_intervention=ri,
        absolute_risk_control=rc,
        comments=comments,
    )

from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any

INTERNAL_CSS = """
<style>
 body { font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;

 .grade-color-VeryLow  { background:#7f1d1d; color:#fff; padding:.2em .6em; border-radius:.4em; font-weight:600; }
 table.sof { width:100%; border-collapse: collapse; font-size: 12.5px; }

 table.sof th { background:#f3f4f6; }
 pre { background:#0b1020; color:#e5e7eb; padding:1em; border-radius:.6em; overflow:auto; }
</style>
"""

@dataclass
class GradeAssRow:
    label: str
    certainty: str
    k: int
    pooled_label: str
    i2_pct: float
    forest_svg: str

@dataclass
class ProjectReportInput:
    project_title: str
    background: str
    methods_pico: str
    prisma_checked_count: int
    grades: list[GradeAssRow] = field(default_factory=list)
    _legend4: bool = False

def _md(inputs: ProjectReportInput) -> str:
    def grade_lines():
        out = []
        for g in inputs.grades:
            out.append(f"### {g.label}\n")
            out.append(f"- **Certainty:** {g.certainty}  ")
            out.append(f"- **Studies (k):** {g.k}  ")
            out.append(f"- **Pooled effect:** {g.pooled_label}  ")
            out.append(f"- **Heterogeneity I²:** {g.i2_pct:.1f}%\n")
            out.append("#### Forest\n")
            out.append((g.forest_svg or "_(Forest plot SVG placeholder)_") + "\n")
        return "\n".join(out) if out else "\n_(No GRADE assessments yet)_\n"
    legend4 = ""
    if inputs._legend4:
        legend4 = (
            "\n**GRADE Certainty Legend:** <span class='grade-color-High'>High</span> · "
            "<span class='grade-color-Moderate'>Moderate</span> · <span class='grade-color-Low'>Low</span> · "
            "<span class='grade-color-VeryLow'>VeryLow</span>\n\n"
        )
    return f"""# {inputs.project_title}

## Background

{inputs.background}

## PICO & Methods

{inputs.methods_pico}

## Results — Summary of Findings & GRADE

{legend4}{grade_lines()}

## PRISMA 2020 Checklist

{inputs.prisma_checked_count} / 27 items checked.

## Discussion

_(Write your discussion here.)_

## Conclusion

_(Write your conclusion here.)_

## References

_(Add references here.)_
"""

def _html(inputs: ProjectReportInput, md_text: str) -> str:
    import html as _html
    body = _html.escape(md_text).replace("\n", "<br/>")
    # Inject forest SVGs (allow <svg> through by replacement marker approach — for simplicity embed inputs.grades[n].forest_svg directly)
    for g in inputs.grades:
        if g.forest_svg and g.forest_svg.strip():
            body = body.replace("_(Forest plot SVG placeholder)_", g.forest_svg, 1)
    legend = ""
    if inputs._legend4:
        legend = (
            "<div class='grade-legend'>"
            "<span class='grade-color-High'>High</span>"
            "<span class='grade-color-Moderate'>Moderate</span>"
            "<span class='grade-color-Low'>Low</span>"
            "<span class='grade-color-VeryLow'>VeryLow</span>"
            "</div>\n"
        )
    return f"""<!doctype html>
<html lang="en"><head><meta charset="utf-8"/><title>{inputs.project_title}</title>
{INTERNAL_CSS}
</head><body>
<h1>{_html.escape(inputs.project_title)}</h1>
{legend}
<article>{body}</article>
</body></html>
"""

def _txt(inputs: ProjectReportInput, md_text: str) -> str:
    # Plain text: strip markdown # chars and _italics_ markers gently
    import re
    t = re.sub(r"^#{1,6}\s+", "", md_text, flags=re.MULTILINE)
    t = re.sub(r"_{1,2}([^_]+)_{1,2}", r"\1", t)
    t = t.replace("`", "")
    return (
        f"{'='*60}\n{inputs.project_title}\n{'='*60}\n\n"
        f"PRISMA 2020 checklist: {inputs.prisma_checked_count}/27\n\n"
        + t
    )

def generate_report_three_formats(inputs: ProjectReportInput) -> tuple[str, str, str]:
    md = _md(inputs)
    html = _html(inputs, md)
    txt = _txt(inputs, md)
    return md, html, txt
```
