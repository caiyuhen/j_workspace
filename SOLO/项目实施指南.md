# 医学智能体协同系统 - 项目实施指南

## 一、项目实施概述

### 1.1 实施策略

采用**敏捷迭代**方式，分三个阶段实施：

| 阶段 | 时间 | 目标 | 交付物 |
|-----|-----|-----|-------|
| Phase 1 (MVP) | 4-6周 | 核心功能可用 | 基础对话 + 代理框架 + LLM对接 |
| Phase 2 | 4-6周 | 功能完善 | 完整代理体系 + Skill集成 + 权限管理 |
| Phase 3 | 4-6周 | 生产就绪 | 部署运维 + 监控告警 + 安全加固 |

### 1.2 技术栈确认

| 层次 | 技术 | 版本 |
|-----|-----|-----|
| 前端 | React + TypeScript | 18.x |
| 后端 | Python FastAPI | 0.100+ |
| 数据库 | PostgreSQL | 15.x |
| 缓存 | Redis | 7.x |
| 大模型服务 | 内置RAG的医学大模型 | 192.168.0.214:8802/chat/ |

```python
import httpx
from typing import List, Dict, AsyncGenerator
from app.config import settings

class LLMService:
    """大模型服务客户端（内置RAG）
    
    大模型服务地址: 192.168.0.214:8802/chat/
    该服务已内置RAG能力，可自动检索医学知识增强生成效果
    """
    
    def __init__(self):
        self.endpoint = settings.LLM_ENDPOINT
        self.model = settings.LLM_MODEL
        self.timeout = settings.LLM_TIMEOUT
    
    async def chat(
        self, 
        messages: List[Dict], 
        temperature: float = 0.7,
        max_tokens: int = 4096
    ) -> Dict:
        """发送聊天请求（自动触发RAG检索增强）"""
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(
                f"{self.endpoint}/chat/completions",
                json={
                    "model": self.model,
                    "messages": messages,
                    "temperature": temperature,
                    "max_tokens": max_tokens
                }
            )
            return response.json()
    
    async def stream_chat(
        self, 
        messages: List[Dict]
    ) -> AsyncGenerator[str, None]:
        """流式聊天（自动RAG增强）"""
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            async with client.stream(
                "POST",
                f"{self.endpoint}/chat/completions",
                json={
                    "model": self.model,
                    "messages": messages,
                    "stream": True
                }
            ) as response:
                async for line in response.aiter_lines():
                    if line.startswith("data: "):
                        yield line[6:]

# 单例
llm_service = LLMService()
```

#### 2.2.4 Step 4: 实现代理基类

创建 `backend/app/agents/base.py`:
```python
from abc import ABC, abstractmethod
from typing import Any, Dict, List, Optional
from enum import Enum
from dataclasses import dataclass
from app.services.llm_service import llm_service

class AgentStatus(Enum):
    IDLE = "idle"
    BUSY = "busy"
    ERROR = "error"

@dataclass
class TaskContext:
    task_id: str
    user_id: str
    conversation_id: str
    input: Any
    metadata: Dict = None

@dataclass
class TaskResult:
    task_id: str
    success: bool
    output: Any
    error: Optional[str] = None

class BaseAgent(ABC):
    """代理基类"""
    
    name: str = "base_agent"
    description: str = ""
    capabilities: List[str] = []
    
    def __init__(self):
        self.status = AgentStatus.IDLE
        self.llm = llm_service
    
    @abstractmethod
    async def execute(self, context: TaskContext) -> TaskResult:
        """执行任务"""
        pass
    
    def can_handle(self, task_type: str) -> bool:
        """判断是否能处理某类型任务"""
        return task_type in self.capabilities
    
    async def call_llm(self, messages: List[Dict], **kwargs) -> Dict:
        """调用大模型（自动RAG增强）"""
        return await self.llm.chat(messages, **kwargs)
```

#### 2.2.5 Step 5: 实现编排代理

创建 `backend/app/agents/orchestrator.py`:
```python
from typing import Dict, List
from app.agents.base import BaseAgent, TaskContext, TaskResult

class OrchestratorAgent(BaseAgent):
    """编排代理 - 系统核心大脑"""
    
    name = "orchestrator"
    description = "任务编排代理，负责分解任务和调度其他代理"
    capabilities = ["task_parsing", "task_decomposition", "agent_scheduling"]
    
    def __init__(self):
        super().__init__()
        self.agents: Dict[str, BaseAgent] = {}
    
    def register_agent(self, agent: BaseAgent):
        """注册代理"""
        self.agents[agent.name] = agent
    
    async def execute(self, context: TaskContext) -> TaskResult:
        """执行编排任务"""
        try:
            # 1. 解析意图
            intent = await self._parse_intent(context.input)
            
            # 2. 分解任务
            subtasks = await self._decompose_task(intent)
            
            # 3. 调度执行
            results = await self._execute_subtasks(subtasks, context)
            
            # 4. 整合结果
            final_result = await self._aggregate_results(results)
            
            return TaskResult(
                task_id=context.task_id,
                success=True,
                output=final_result
            )
        except Exception as e:
            return TaskResult(
                task_id=context.task_id,
                success=False,
                output=None,
                error=str(e)
            )
    
    async def _parse_intent(self, user_input: str) -> Dict:
        """解析用户意图"""
        response = await self.call_llm([
            {"role": "system", "content": "你是一个意图识别专家，分析用户输入并识别意图。"},
            {"role": "user", "content": f"分析以下输入的意图：{user_input}"}
        ])
        return response
    
    async def _decompose_task(self, intent: Dict) -> List[Dict]:
        """分解任务"""
        response = await self.call_llm([
            {"role": "system", "content": "你是一个任务分解专家。"},
            {"role": "user", "content": f"将以下意图分解为子任务：{intent}"}
        ])
        return response.get("subtasks", [])
    
    async def _execute_subtasks(self, subtasks: List[Dict], context: TaskContext) -> List[TaskResult]:
        """执行子任务"""
        results = []
        for subtask in subtasks:
            task_type = subtask.get("type")
            for agent in self.agents.values():
                if agent.can_handle(task_type):
                    result = await agent.execute(context)
                    results.append(result)
                    break
        return results
    
    async def _aggregate_results(self, results: List[TaskResult]) -> Dict:
        """整合结果"""
        successful = [r.output for r in results if r.success]
        response = await self.call_llm([
            {"role": "system", "content": "你是一个结果整合专家。"},
            {"role": "user", "content": f"整合以下结果：{successful}"}
        ])
        return response
```

```python
from app.agents.base import BaseAgent, TaskContext, TaskResult

class DiagnosisAgent(BaseAgent):
    """诊断代理 - 临床诊断辅助"""
    
    name = "diagnosis"
    description = "临床诊断辅助代理，提供病历分析和诊断建议"
    capabilities = ["medical_record_analysis", "diagnosis_suggestion", "differential_diagnosis"]
    
    async def execute(self, context: TaskContext) -> TaskResult:
        """执行诊断任务"""
        task_type = context.metadata.get("task_type", "diagnosis_suggestion")
        
        if task_type == "medical_record_analysis":
            return await self._analyze_record(context)
        elif task_type == "diagnosis_suggestion":
            return await self._suggest_diagnosis(context)
        else:
            return TaskResult(
                task_id=context.task_id,
                success=False,
                output=None,
                error=f"Unknown task type: {task_type}"
            )
    
    async def _analyze_record(self, context: TaskContext) -> TaskResult:
        """分析病历"""
        record = context.input.get("record", "")
        response = await self.call_llm([
            {"role": "system", "content": "你是一位资深临床医生，擅长病历分析。"},
            {"role": "user", "content": f"请分析以下病历：\n{record}"}
        ])
        return TaskResult(task_id=context.task_id, success=True, output=response)
    
    async def _suggest_diagnosis(self, context: TaskContext) -> TaskResult:
        """诊断建议"""
        symptoms = context.input.get("symptoms", [])
        response = await self.call_llm([
            {"role": "system", "content": "你是一位资深临床医生，擅长诊断推理。"},
            {"role": "user", "content": f"根据以下症状给出诊断建议：{symptoms}"}
        ])
        return TaskResult(task_id=context.task_id, success=True, output=response)
```

#### 2.2.7 Step 7: 实现代理注册中心

创建 `backend/app/agents/registry.py`:
```python
from typing import Dict, List, Optional
from app.agents.base import BaseAgent
from app.agents.orchestrator import OrchestratorAgent
from app.agents.diagnosis import DiagnosisAgent

class AgentRegistry:
    """代理注册中心"""
    
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
        self._initialized = True
        self.agents: Dict[str, BaseAgent] = {}
        self.orchestrator: Optional[OrchestratorAgent] = None
        self._setup_agents()
    
    def _setup_agents(self):
        """初始化代理"""
        # 创建编排代理
        self.orchestrator = OrchestratorAgent()
        self.agents["orchestrator"] = self.orchestrator
        
        # 创建专业代理
        diagnosis_agent = DiagnosisAgent()
        self.agents["diagnosis"] = diagnosis_agent
        
        # 注册到编排代理
        self.orchestrator.register_agent(diagnosis_agent)
    
    def get_agent(self, name: str) -> Optional[BaseAgent]:
        """获取代理"""
        return self.agents.get(name)
    
    def list_agents(self) -> List[str]:
        """列出所有代理"""
        return list(self.agents.keys())

# 全局单例
agent_registry = AgentRegistry()
```

#### 2.2.8 Step 8: 实现API接口

创建 `backend/app/api/v1/conversations.py`:
```python
from fastapi import APIRouter, Depends
from typing import List
from pydantic import BaseModel
from app.agents.registry import agent_registry
from app.agents.base import TaskContext
import uuid

router = APIRouter()

class MessageCreate(BaseModel):
    content: str
    input_type: str = "text"

class MessageResponse(BaseModel):
    id: str
    role: str
    content: str
    agent: str = None

@router.post("/{conversation_id}/messages")
async def send_message(
    conversation_id: str,
    message: MessageCreate
):
    """发送消息"""
    # 创建任务上下文
    context = TaskContext(
        task_id=str(uuid.uuid4()),
        user_id="user_001",  # 从JWT获取
        conversation_id=conversation_id,
        input=message.content,
        metadata={"task_type": "diagnosis_suggestion"}
    )
    
    # 通过编排代理处理
    orchestrator = agent_registry.orchestrator
    result = await orchestrator.execute(context)
    
    return {
        "message_id": str(uuid.uuid4()),
        "content": result.output if result.success else result.error,
        "success": result.success
    }
```
